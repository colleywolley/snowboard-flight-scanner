1) route_key = f"{orig}-{dest}" if route_key not in best_by_route or total_price < best_by_route[route_key]["price_total_2_adults"]: best_by_route[route_key] = { "orig": orig, "dest": dest, "price_total_2_adults": total_price, "price_per_person": round(total_price / 2), "airline": airline, "stops": stops, } return best_by_route def send_webhook(webhook_url: str, message: str) -> None: if not webhook_url: print(message) return if "ntfy.sh/" in webhook_url: req = urllib.request.Request( webhook_url, data=message.encode("utf-8"), headers={"Title": "Snowboarding Flight Price Alert", "User-Agent": "SnowboardFlightScanner/1.0"}, method="POST", ) else: payload = json.dumps({"content": message, "text": message}).encode("utf-8") req = urllib.request.Request( webhook_url, data=payload, headers={"Content-Type": "application/json", "User-Agent": "SnowboardFlightScanner/1.0"}, method="POST", ) urllib.request.urlopen(req, timeout=10) def main() -> None: api_key = os.environ.get("SERPAPI_KEY", "") webhook_url = os.environ.get("ALERT_WEBHOOK_URL", "") if not api_key: print("Set SERPAPI_KEY environment variable to query live Google Flights data.") return history = json.loads(STATE_FILE.read_text()) if STATE_FILE.exists() else {} alerts = [] for dep, ret, label in WEEKENDS: best_by_route = fetch_weekend_fares(api_key, dep, ret) for route_key, result in best_by_route.items(): orig = result["orig"] dest = result["dest"] key = f"{route_key}_{dep}_{ret}" pp = result["price_per_person"] prev = history.get(key, {}).get("price_per_person") threshold = LOW_PRICE_THRESHOLDS.get(dest, 300) stops_str = "Nonstop" if result["stops"] == 0 else f"{result['stops']} stop" if prev is not None and pp != prev: diff = pp - prev direction = f"DROPPED by ${abs(diff)}" if diff < 0 else f"rose by ${diff}" alerts.append( f"📊 **{orig} → {dest}** ({label}: {dep} to {ret}) {direction}: " f"now **${pp}/person** (${result['price_total_2_adults']} for 2) on {result['airline']} ({stops_str})" ) elif pp <= threshold: alerts.append( f"🔥 **LOW PRICE ALERT: {orig} → {dest}** ({label}: {dep} to {ret}) is **${pp}/person** " f"(below ${threshold} target!) on {result['airline']} ({stops_str})" ) history[key] = result STATE_FILE.write_text(json.dumps(history, indent=2)) if alerts: send_webhook(webhook_url, "🏂 **Snowboarding Flight Price Update**\n" + "\n".join(alerts)) else: print("Scan complete.
